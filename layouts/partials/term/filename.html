{{- /* 8.3 file name for a page */ -}} {{- $f := "" -}} {{- if .Params.file }}{{ $f = .Params.file }} {{- else if eq .Section "issues" }}{{ $f = printf "ISSUE%03d.TXT" (int (.Params.issue | default 0)) }} {{- else if eq .Section "columns" }}{{ $f = printf "OP%s.TXT" (.Date.Format "060102") }} {{- else if eq .Layout "about" }}{{ $f = "ABOUT.TXT" }} {{- else if eq .Layout "subscribe" }}{{ $f = "SUBSCRIB.EXE" }} {{- else }}{{ $f = printf "%s.TXT" (upper (substr (.File.ContentBaseName | urlize) 0 8)) }} {{- end -}} {{- return $f -}}
